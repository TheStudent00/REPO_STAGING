1. c/op_3 (!a, a:f32 -- packed EQ mask truthy)
    unit: c/op_3   source: canon19_text
    xorps %xmm3,%xmm3     zero %xmm3
    movaps %xmm3,%xmm2    %xmm2 = %xmm3
    cmpeqss %xmm0,%xmm2   %xmm2 = mask_equal_f32(%xmm2, %xmm0)
    movd %xmm2,%eax       %eax = %xmm2
    and $1,%al            %al = %al & $1
    movzbl %al,%eax       %eax = zero_extend(%al)
    ret                   return eax

3. c/op_303 (a || b, a:f32 b:f32 -- two NE masks ORed)
    unit: c/op_303   source: canon19_text
    xorps %xmm3,%xmm3      zero %xmm3
    movaps %xmm0,%xmm2     %xmm2 = %xmm0
    cmpneqss %xmm3,%xmm2   %xmm2 = mask_equal_f32(%xmm2, %xmm3)
    movaps %xmm1,%xmm4     %xmm4 = %xmm1
    cmpneqss %xmm3,%xmm4   %xmm4 = mask_equal_f32(%xmm4, %xmm3)
    orps %xmm4,%xmm2       %xmm2 = %xmm2 | %xmm4
    movd %xmm2,%eax        %eax = %xmm2
    and $1,%al             %al = %al & $1
    movzbl %al,%eax        %eax = zero_extend(%al)
    ret                    return eax
