1. swift/op_296 (a < b, a:Int32 b:UInt64 -- the sign-test bug fix)
    unit: swift/op_296   source: canon18_text
    mov %rdi,%rax        %rax = %rdi
    shr $8,%rax          %rax = shift_right_zeros(%rax, $8)
    and $16777215,%eax   %eax = %eax & $16777215
    shl $7,%eax          %eax = shift_left(%eax, $7)
    mov $0,%r10b         %r10b = $0
    movzbl %r10b,%r10d   %r10d = zero_extend(%r10b)
    or %r10d,%eax        %eax = %eax | %r10d
    shl $1,%eax          %eax = shift_left(%eax, $1)
    mov $0,%r11          %r11 = $0
    mov %rdi,%r9         %r9 = %rdi
    and $-1,%r9d         %r9d = %r9d & $-1
    movsxd %r9d,%r9      %r9 = sign_extend(%r9d)
    cmp %r9,%r11         flags = compare(%r11, %r9)
    setg %r10b           %r10b = flags.signed_greater
    mov %rsi,%r9         %r9 = %rsi
    mov %rdi,%r8         %r8 = %rdi
    and $-1,%r8d         %r8d = %r8d & $-1
    movsxd %r8d,%r8      %r8 = sign_extend(%r8d)
    cmp %r8,%r9          flags = compare(%r9, %r8)
    seta %r11b           %r11b = flags.above
    or %r11b,%r10b       %r10b = %r10b | %r11b
    movzbl %r10b,%r10d   %r10d = zero_extend(%r10b)
    or %r10d,%eax        %eax = %eax | %r10d
    movsxd %eax,%rax     %rax = sign_extend(%eax)
    ret                  return eax
